## 给某个用户授予某个数据库权限
```sql
USE [库名];
GO

-- 删除可能存在的孤立用户
IF EXISTS (SELECT 1 FROM sys.database_principals WHERE name = N'用户名')
BEGIN
    DROP USER [用户名];
END
GO

-- 创建数据库用户，映射到已有登录名
CREATE USER [用户名] FOR LOGIN [用户名] WITH DEFAULT_SCHEMA=[dbo];
GO

-- 给足权限（整个数据库所有权限）
ALTER ROLE [db_owner] ADD MEMBER [用户名];
GO
```


## 给数据库改所有者


例：所有者改为 sa
```sql
USE [数据库名];
GO
ALTER AUTHORIZATION ON DATABASE::[数据库名] TO [sa];
GO
```

验证一下
```sql
SELECT name, suser_sname(owner_sid) AS owner
FROM sys.databases
WHERE name = '数据库名';
```



## 创建link

```sql
EXEC sp_addlinkedserver 
    @server = 'rdmLink',            -- 链接服务器名称
    @srvproduct = '',               -- SQL Server 可留空
    @provider = 'SQLNCLI',          -- 或者直接省略，默认就是 SQL Server Native Client
    @datasrc = '192.168.35.43';     -- 目标 SQL Server 地址

-- 配置登录映射
EXEC sp_addlinkedsrvlogin 
    @rmtsrvname = 'rdmLink',
    @useself = 'False',
    @locallogin = NULL,
    @rmtuser = 'sa',
    @rmtpassword = 'P@ssw0rd';
```
