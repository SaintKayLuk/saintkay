# 删除流水线任务

因为gitlab没有一键删除所有流水线或批量删除，则可以通过脚本来删除

```sh
#!/bin/bash

# gitlab地址
GITLAB_URL="https://gitlab.wecharmer.com"
# 项目名，组名/项目名，当然也可以循环写
PROJECT_PATH="wms%2Ffrontend%2Fwecharmer-wms-vue"
# token
PRIVATE_TOKEN="glpat-PYLd2eZgWgxmcB8aBiZA"

# 获取项目ID
PROJECT_ID=$(curl --silent --header "PRIVATE-TOKEN: $PRIVATE_TOKEN" "$GITLAB_URL/api/v4/projects/$(echo $PROJECT_PATH | sed 's/\//%2F/g')" | jq '.id')

echo "Project ID: $PROJECT_ID"

# 获取所有 pipeline id，并删除
page=1
while true; do
  pipelines=$(curl --silent --header "PRIVATE-TOKEN: $PRIVATE_TOKEN" "$GITLAB_URL/api/v4/projects/$PROJECT_ID/pipelines?per_page=100&page=$page" | jq '.[].id')

  if [ -z "$pipelines" ]; then
    echo "All pipelines deleted or no more pipelines."
    break
  fi

  for pipeline_id in $pipelines; do
    echo "Deleting pipeline ID: $pipeline_id"
    curl --silent --request DELETE --header "PRIVATE-TOKEN: $PRIVATE_TOKEN" "$GITLAB_URL/api/v4/projects/$PROJECT_ID/pipelines/$pipeline_id"
  done

  ((page++))
done

```
